def dcg_at_k(relevances):
    rel = np.asarray(relevances, dtype=float)
    if rel.size == 0:
        return 0.0
    discounts = 1.0 / np.log2(np.arange(2, rel.size + 2))
    return float(np.sum(rel * discounts))

def ndcg_at_k(pred_items, true_items, k=10):
    pred_items = list(pred_items)[:k]
    true_items = set(true_items)
    if not true_items:
        return np.nan
    gains = [1.0 if i in true_items else 0.0 for i in pred_items]
    ideal = [1.0] * min(len(true_items), k)
    idcg = dcg_at_k(ideal)
    return 0.0 if idcg == 0 else dcg_at_k(gains) / idcg

def recall_at_k(pred_items, true_items, k=10):
    pred_items = set(list(pred_items)[:k])
    true_items = set(true_items)
    if not true_items:
        return np.nan
    return len(pred_items & true_items) / len(true_items)

        except TypeError:
            rec_u = algo.recommend(int(u), n)

def sample_eval_pairs(train_df, test_df, item_pool, n_neg=5, seed=0):
    rng = np.random.default_rng(seed)
    pairs, labels = [], []
    train_map = train_df.groupby('user')['item'].apply(set).to_dict()
    test_map = test_df.groupby('user')['item'].apply(set).to_dict()
    item_pool = np.asarray(item_pool)
    item_pool_set = set(item_pool.tolist())
    for u, pos_items in test_map.items():
        pos_items = set(pos_items)
        seen = train_map.get(u, set()).union(pos_items)
        candidates = np.array(sorted(item_pool_set.difference(seen)), dtype=np.int64)
        for i in pos_items:
            pairs.append((u, i))
            labels.append(1.0)
        if len(candidates):
            k = min(len(pos_items) * n_neg, len(candidates))
            negs = rng.choice(candidates, size=k, replace=False)
            for i in negs:
                pairs.append((u, int(i)))
                labels.append(0.0)
    return pd.DataFrame(pairs, columns=['user', 'item']), np.asarray(labels, dtype=np.float32)


def log_loss_from_scores(scores, labels):
    probs = sigmoid(np.asarray(scores, dtype=np.float64))
    probs = np.clip(probs, 1e-8, 1 - 1e-8)
    labels = np.asarray(labels, dtype=np.float64)
    return float(-(labels * np.log(probs) + (1 - labels) * np.log(1 - probs)).mean())


def safe_recommend(algo, users, n, candidate_items=None):
    rec_frames = []
    for u in users:
        try:
            if candidate_items is None:
                rec = algo.recommend(int(u), n)
            else:
                rec = algo.recommend(int(u), n, candidates=candidate_items)
        except TypeError:
            rec = algo.recommend(int(u), n)
        if rec is None:
            continue
        if not isinstance(rec, pd.DataFrame):
            rec = pd.DataFrame(rec)
        if len(rec) == 0:
            continue
        if 'user' not in rec.columns:
            rec = rec.copy()
            rec['user'] = int(u)
        rec_frames.append(rec)
    if rec_frames:
        return pd.concat(rec_frames, ignore_index=True)
    return pd.DataFrame(columns=['user', 'item', 'score', 'rank'])


def safe_predict(algo, pairs_df, fill_value=-20.0):
    scores = np.empty(len(pairs_df), dtype=np.float64)
    for idx, row in enumerate(pairs_df.itertuples(index=False)):
        try:
            pred = algo.predict_for_user(int(row.user), [int(row.item)])
            val = pred.iloc[0] if hasattr(pred, 'iloc') else list(pred)[0]
            if pd.isna(val):
                val = fill_value
        except Exception:
            val = fill_value
        scores[idx] = float(val)
    return scores

def sample_eval_pairs(train_df, test_df, item_pool, n_neg=5, seed=0):
    rng = np.random.default_rng(seed)
    pairs, labels = [], []
    train_map = train_df.groupby('user')['item'].apply(set).to_dict()
    test_map = test_df.groupby('user')['item'].apply(set).to_dict()
    item_pool = np.asarray(item_pool, dtype=np.int64)
    item_pool_set = set(item_pool.tolist())
    for u, pos_items in test_map.items():
        pos_items = set(pos_items)
        seen = train_map.get(u, set()).union(pos_items)
        candidates = np.array(sorted(item_pool_set.difference(seen)), dtype=np.int64)
        for i in pos_items:
            pairs.append((int(u), int(i)))
            labels.append(1.0)
        if len(candidates):
            k = min(len(pos_items) * n_neg, len(candidates))
            negs = rng.choice(candidates, size=k, replace=False)
            for i in negs:
                pairs.append((int(u), int(i)))
                labels.append(0.0)
    return pd.DataFrame(pairs, columns=['user', 'item']), np.asarray(labels, dtype=np.float32)


def log_loss_from_scores(scores, labels):
    probs = sigmoid(scores)
    probs = np.clip(probs, 1e-8, 1 - 1e-8)
    labels = np.asarray(labels, dtype=np.float64)
    return float(-(labels * np.log(probs) + (1 - labels) * np.log(1 - probs)).mean())


def _get_scorer(obj):
    for attr in ('predictor', 'scorer', 'algorithm'):

    return obj


def score_user_items(algo, user, items, fill_value=-20.0):
    items = [int(i) for i in items]
    if len(items) == 0:
        return pd.Series(dtype=np.float64)
    scorer = _get_scorer(algo)

    except Exception:
        try:
            s = scorer.predict_for_user(int(user), items)
        except Exception:
            return pd.Series(fill_value, index=pd.Index(items, name='item'), dtype=np.float64)
    if isinstance(s, pd.Series):
        out = s.reindex(items)
    else:
        try:
            out = pd.Series(np.asarray(s, dtype=np.float64), index=pd.Index(items, name='item'))
        except Exception:
            out = pd.Series(fill_value, index=pd.Index(items, name='item'), dtype=np.float64)

    return out


def safe_recommend(algo, users, n, train_map, item_pool):
    rec_frames = []
    item_pool = np.asarray(item_pool, dtype=np.int64)
    for u in users:
        u = int(u)
        rec = None
        try:
            rec = algo.recommend(u, n)
        except Exception:
            rec = None
        if rec is None or len(rec) == 0:
            seen = train_map.get(u, set())
            candidates = [int(i) for i in item_pool if int(i) not in seen]
            scores = score_user_items(algo, u, candidates)
            if len(scores) == 0:
                continue
            scores = scores.sort_values(ascending=False).head(n)
            rec = pd.DataFrame({
                'item': scores.index.to_numpy(dtype=np.int64),
                'score': scores.to_numpy(dtype=np.float64),
                'rank': np.arange(1, len(scores) + 1, dtype=np.int64),
                'user': u
            })
        else:
            if not isinstance(rec, pd.DataFrame):
                rec = pd.DataFrame(rec)
            rec = rec.copy()
            if 'user' not in rec.columns:
                rec['user'] = u
            if 'rank' not in rec.columns:
                rec['rank'] = np.arange(1, len(rec) + 1, dtype=np.int64)
        rec_frames.append(rec[['user', 'item', 'score', 'rank']] if 'score' in rec.columns else rec[['user', 'item', 'rank']])
    if rec_frames:
        return pd.concat(rec_frames, ignore_index=True)
    return pd.DataFrame(columns=['user', 'item', 'score', 'rank'])
