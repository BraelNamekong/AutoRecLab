import numpy as np
import pandas as pd

        'ground_truth': [],

    }

# Load MovieLens100K u.data
cols = ['user', 'item', 'rating', 'timestamp']
ratings = pd.read_csv('u.data', sep='\t', names=cols, engine='python')
ratings = ratings[['user', 'item', 'rating', 'timestamp']].copy()
ratings['rating'] = 1.0  # implicit normalization / binarization

# Fixed user-level holdout split so only model seed changes across runs
rng = np.random.RandomState(2024)
test_idx = []
for user, grp in ratings.groupby('user'):
    idx = grp.index.to_numpy()
    if len(idx) >= 2:
        test_idx.append(rng.choice(idx, size=1, replace=False)[0])

test = ratings.loc[sorted(test_idx), ['user', 'item', 'rating']].copy()
train = ratings.drop(index=test_idx)[['user', 'item', 'rating']].copy()

# Candidates exclude training items; test contains one relevant item per user
users = test['user'].unique()
all_items = pd.Index(ratings['item'].unique())
train_items = train.groupby('user')['item'].apply(set).to_dict()

def make_candidates(user_ids):
    rows = []
    for u in user_ids:
        seen = train_items.get(u, set())
        cand = all_items[~all_items.isin(list(seen))]
        rows.append(pd.DataFrame({'user': u, 'item': cand}))
    return pd.concat(rows, ignore_index=True)

candidates = make_candidates(users)
truth = test[['user', 'item']].copy()
truth['rating'] = 1.0

seeds = [1, 7, 13, 21, 42, 84, 126, 168, 210, 252]
run_rows = []

for run_id, seed in enumerate(seeds, start=1):
    algo = ImplicitMF(features=40, iterations=15, reg=0.1, weight=40, method='cg', rng_spec=seed)
    algo.fit(train)

    recs = batch.recommend(algo, users, 20, candidates=candidates)
    recs = recs[['user', 'item', 'rank', 'score']].copy()

    rla = RecListAnalysis()
    rla.add_metric('ndcg')
    rla.add_metric('recall')
    metrics = rla.compute(recs, truth)
    mean_ndcg = float(metrics['ndcg'].mean())
    mean_recall = float(metrics['recall'].mean())

    ts = pd.Timestamp.utcnow().isoformat()
    experiment_data['ml100k']['metrics']['val'].append({
        'run': run_id,
        'seed': seed,
        'ndcg@20': mean_ndcg,
        'recall@20': mean_recall,
        'timestamp': ts
    })
    experiment_data['ml100k']['losses']['val'].append({
        'run': run_id,
        'seed': seed,
        'validation_loss': np.nan,
        'timestamp': ts
    })
    experiment_data['ml100k']['predictions'].append(recs.head(200).to_dict('records'))
    experiment_data['ml100k']['ground_truth'].append(truth.head(200).to_dict('records'))
    experiment_data['ml100k']['timestamps'].append(ts)
    experiment_data['ml100k']['seeds'].append(seed)

    run_rows.append({'seed': seed, 'ndcg@20': mean_ndcg, 'recall@20': mean_recall})
    print(f'Epoch {run_id}: validation_loss = {float("nan"):.4f}')
    print(f'Run {run_id} seed={seed}: NDCG@20={mean_ndcg:.4f}, Recall@20={mean_recall:.4f}')

os.makedirs(working_dir, exist_ok=True)

import time
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from lenskit import batch, topn
from lenskit.algorithms.als import ImplicitMF
from lenskit.algorithms import Recommender
from sklearn.model_selection import train_test_split

# -----------------------------
# Data + experiment containers
# -----------------------------
experiment_data = {
    'movielens100k': {
        'metrics': {'train': [], 'val': []},
        'losses': {'train': [], 'val': []},
        'predictions': [],
        'ground_truth': [],
        'timestamps': [],
        'seeds': [],
        'per_run': []
    }
}

# -----------------------------
# Load MovieLens100K
# -----------------------------
ratings = pd.read_csv(
    'u.data', sep='\t', header=None,
    names=['user', 'item', 'rating', 'timestamp']
)

# Implicit-feedback view: every observed interaction is positive.
# Keep a normalized confidence/weight signal in [0,1] for model input.
ratings['rating'] = ratings['rating'].astype(float)
ratings['weight'] = ratings['rating'] / ratings['rating'].max()
implicit = ratings[['user', 'item', 'weight']].rename(columns={'weight': 'rating'})

# Fixed split so only the model seed changes.
train_df, test_df = train_test_split(
    implicit, test_size=0.2, random_state=2024, stratify=implicit['user']
)
train_df = train_df.reset_index(drop=True)
test_df = test_df.reset_index(drop=True)

# Restrict evaluation to users with test interactions and at least one train interaction.
test_users = np.intersect1d(test_df['user'].unique(), train_df['user'].unique())
test_df = test_df[test_df['user'].isin(test_users)].copy()

# LensKit truth format
truth = test_df.groupby('user')['item'].apply(list)
experiment_data['movielens100k']['ground_truth'] = truth.to_dict()

# -----------------------------
# Experiment settings
# -----------------------------
seeds = [1, 7, 21, 42, 84, 168, 256, 512, 1024, 2048]
K = 20
features = 40
iterations = 15
reg = 0.1
weight = 40

# -----------------------------
# Run repeated training/eval
# -----------------------------
ndcg_vals, recall_vals = [], []
all_preds = {}

for run_idx, seed in enumerate(seeds, start=1):
    ts = time.time()
    algo = ImplicitMF(features=features, iterations=iterations, reg=reg, weight=weight, progress=False, method='cg', rng_spec=seed)
    model = Recommender.adapt(algo)
    model.fit(train_df)

    recs = batch.recommend(model, test_users, K)
    recs = recs.sort_values(['user', 'rank'])

    rla = topn.RecListAnalysis()
    rla.add_metric(topn.ndcg)
    rla.add_metric(topn.recall)
    results = rla.compute(recs, truth)

    ndcg = float(results['ndcg'].mean())
    recall = float(results['recall'].mean())
    val_loss = float('nan')

    print(f'Epoch {run_idx}: validation_loss = {val_loss:.4f}')
    print(f'Run {run_idx:02d} | seed={seed} | NDCG@{K}={ndcg:.4f} | Recall@{K}={recall:.4f}')

    ndcg_vals.append(ndcg)
    recall_vals.append(recall)
    all_preds[int(seed)] = recs[['user', 'item', 'rank', 'score']].to_records(index=False)

    experiment_data['movielens100k']['metrics']['val'].append({
        'run': run_idx,
        'seed': seed,
        'ndcg': ndcg,
        'recall': recall,
        'timestamp': ts
    })
    experiment_data['movielens100k']['losses']['val'].append({
        'run': run_idx,
        'seed': seed,
        'validation_loss': val_loss,
        'timestamp': ts
    })
    experiment_data['movielens100k']['timestamps'].append(ts)
    experiment_data['movielens100k']['seeds'].append(seed)
    experiment_data['movielens100k']['per_run'].append({
        'run': run_idx,
        'seed': seed,
        'ndcg': ndcg,
        'recall': recall
    })

# -----------------------------
# Summaries
# -----------------------------
ndcg_arr = np.array(ndcg_vals, dtype=float)
recall_arr = np.array(recall_vals, dtype=float)
seed_arr = np.array(seeds, dtype=int)

print('\nSummary across seeds')
print(f'NDCG@{K}:   mean={ndcg_arr.mean():.4f}, std={ndcg_arr.std(ddof=1):.4f}, min={ndcg_arr.min():.4f}, max={ndcg_arr.max():.4f}')
print(f'Recall@{K}: mean={recall_arr.mean():.4f}, std={recall_arr.std(ddof=1):.4f}, min={recall_arr.min():.4f}, max={recall_arr.max():.4f}')

# -----------------------------
# Save arrays and dict
# -----------------------------
np.save(os.path.join(working_dir, 'ml100k_seeds.npy'), seed_arr)
np.save(os.path.join(working_dir, 'ml100k_ndcg.npy'), ndcg_arr)
np.save(os.path.join(working_dir, 'ml100k_recall.npy'), recall_arr)
np.save(os.path.join(working_dir, 'experiment_data.npy'), experiment_data, allow_pickle=True)
np.savez_compressed(os.path.join(working_dir, 'ml100k_predictions.npz'), **{f'seed_{k}': v for k, v in all_preds.items()})

# -----------------------------
# Visualizations
# -----------------------------
plt.figure(figsize=(8, 4))
plt.plot(seed_arr, ndcg_arr, marker='o', label=f'NDCG@{K}')
plt.plot(seed_arr, recall_arr, marker='s', label=f'Recall@{K}')
plt.xlabel('Random seed')
plt.ylabel('Metric value')
plt.title('MovieLens100K ImplicitMF metrics across random seeds')
plt.legend()
plt.tight_layout()
plt.savefig(os.path.join(working_dir, 'movielens100k_metrics_across_seeds.png'), dpi=150)
plt.close()

plt.figure(figsize=(8, 4))
plt.boxplot([ndcg_arr, recall_arr], labels=[f'NDCG@{K}', f'Recall@{K}'])
plt.ylabel('Metric value')
plt.title('MovieLens100K metric variability across seeds')
plt.tight_layout()
plt.savefig(os.path.join(working_dir, 'movielens100k_metric_variability_boxplot.png'), dpi=150)
plt.close()

def recommend_serial(model, users, n):
    frames = []
    for user in users:
        try:
            rec = model.recommend(user, n)
        except TypeError:
            rec = model.recommend(user, n=n)
        if rec is None or len(rec) == 0:
            continue
        rec = rec.copy()
        if 'user' not in rec.columns:
            rec['user'] = user
        if 'rank' not in rec.columns:
            rec['rank'] = np.arange(1, len(rec) + 1)
        cols = [c for c in ['user', 'item', 'rank', 'score'] if c in rec.columns]
        frames.append(rec[cols])
    if not frames:
        return pd.DataFrame(columns=['user', 'item', 'rank', 'score'])
    recs = pd.concat(frames, ignore_index=True)
    if 'rank' not in recs.columns:
        recs['rank'] = recs.groupby('user').cumcount() + 1
    return recs.sort_values(['user', 'rank']).reset_index(drop=True)
