def stratified_split(df, test_frac, seed):

    train_chunks = []

    for user_id, group in df.groupby('user'):

# Simple evaluation: Recall@K and NDCG@K aggregated over users with relevant test items
def evaluate_rankings(pred_df, test_df, K):
    # pred_df columns: user, item, score
    if pred_df is None or pred_df.empty:
        return 0.0, 0.0
    # Build per-user score map
    pred_map = {}
    for _, row in pred_df.iterrows():
        u = int(row['user'])
        i = int(row['item'])
        s = float(row['score'])
        pred_map[(u, i)] = s
    # Ground-truth relevance from test_df (rating >= 4 considered relevant)
    test_df = test_df.copy()
    test_df['relevant'] = (test_df['rating'] >= 4.0).astype(int)

    total_relevance = 0
    total_relevant_in_topk = 0
    total_dcg = 0.0
    total_idcg = 0.0
    user_count = 0

    for u, group in test_df.groupby('user'):
        user_items = group['item'].tolist()
        if len(user_items) == 0:
            continue
        # relevant flag per item in test
        rel_dict = {}
        for _, r in group.iterrows():
            rel_dict[int(r['item'])] = int(r['relevant'])
        # Build scores for these items from pred_map; unseen get very low score
        scores = [(i, pred_map.get((int(u), i), -1e6)) for i in user_items]
        scores.sort(key=lambda x: x[1], reverse=True)
        ranked = [i for i, _ in scores[:K]]
        # Compute recall and DCG/IDCG for this user
        # total relevant in test for this user
        R = sum(rel_dict.values())
        total_relevance += R
        # Recall: among top-K, how many relevant items appear
        topk_rel = sum(rel_dict.get(i, 0) for i in ranked)
        total_relevant_in_topk += topk_rel
        # DCG and IDCG
        dcg = 0.0
        for idx, item in enumerate(ranked, start=1):
            rel = rel_dict.get(item, 0)
            if rel > 0:
                dcg += rel / np.log2(idx + 1)
        total_dcg += dcg
        idcg = 0.0
        for i in range(1, min(R, K) + 1):
            idcg += 1.0 / np.log2(i + 1)
        total_idcg += idcg
        user_count += 1

    recall = (total_relevant_in_topk / total_relevance) if total_relevance > 0 and user_count>0 else 0.0
    ndcg = (total_dcg / total_idcg) if total_idcg > 0 else 0.0
    return recall, ndcg

class BaselineModel:
    def __init__(self, train_df):
        self.global_mean = float(train_df['rating'].mean()) if not train_df.empty else 0.0
        user_mean = train_df.groupby('user')['rating'].mean()
        item_mean = train_df.groupby('item')['rating'].mean()
        self.user_bias = (user_mean - self.global_mean).to_dict()
        self.item_bias = (item_mean - self.global_mean).to_dict()
    def predict(self, user, item):
        ub = self.user_bias.get(user, 0.0)
        ib = self.item_bias.get(item, 0.0)
        return self.global_mean + ub + ib
    def predict_df(self, df):
        return df.assign(score=df.apply(lambda r: self.predict(int(r['user']), int(r['item'])), axis=1))

# Stratified per-user split
def stratified_split(df, test_frac, seed):
    rng = np.random.RandomState(seed)
    train_chunks = []
    test_chunks = []
    for user_id, group in df.groupby('user'):
        if len(group) < 2:
            train_chunks.append(group)
        else:
            n_test = max(1, int(len(group) * test_frac))
            idx = group.index.values
            test_idx = rng.choice(idx, size=n_test, replace=False)
            train_idx = [i for i in idx if i not in set(test_idx)]
            test_chunks.append(group.loc[test_idx])
            train_chunks.append(group.loc[train_idx])
    train_df = pd.concat(train_chunks) if train_chunks else pd.DataFrame(columns=df.columns)
    test_df = pd.concat(test_chunks) if test_chunks else pd.DataFrame(columns=df.columns)
    return train_df, test_df

# Simple evaluation: Recall@K and NDCG@K aggregated over users with relevant test items
def evaluate_rankings(pred_df, test_df, K):
    if pred_df is None or pred_df.empty:
        return 0.0, 0.0
    pred_map = {}
    for _, row in pred_df.iterrows():
        u = int(row['user']); i = int(row['item']); s = float(row['score'])
        pred_map[(u, i)] = s
    test_df = test_df.copy()
    test_df['relevant'] = (test_df['rating'] >= 4.0).astype(int)

    total_relevance = 0
    total_relevant_in_topk = 0
    total_dcg = 0.0
    total_idcg = 0.0
    user_count = 0

    for u, group in test_df.groupby('user'):
        user_items = group['item'].tolist()
        if len(user_items) == 0:
            continue
        rel_dict = {}
        for _, r in group.iterrows():
            rel_dict[int(r['item'])] = int(r['relevant'])
        scores = [(i, pred_map.get((int(u), i), -1e6)) for i in user_items]
        scores.sort(key=lambda x: x[1], reverse=True)
        ranked = [i for i, _ in scores[:K]]
        R = sum(rel_dict.values())
        total_relevance += R
        topk_rel = sum(rel_dict.get(i, 0) for i in ranked)
        total_relevant_in_topk += topk_rel
        dcg = 0.0
        for idx, item in enumerate(ranked, start=1):
            rel = rel_dict.get(item, 0)
            if rel > 0:
                dcg += rel / np.log2(idx + 1)
        total_dcg += dcg
        idcg = 0.0
        for i in range(1, min(R, K) + 1):
            idcg += 1.0 / np.log2(i + 1)
        total_idcg += idcg
        user_count += 1

    recall = (total_relevant_in_topk / total_relevance) if total_relevance > 0 and user_count>0 else 0.0
    ndcg = (total_dcg / total_idcg) if total_idcg > 0 else 0.0
    return recall, ndcg

# Compute a simple validation loss: MSE on test ratings vs predicted scores
def compute_validation_loss(pred_df, test_df):
    if pred_df is None or pred_df.empty:
        return None
    merge = pred_df.merge(test_df[['user','item','rating']], on=['user','item'], how='left')
    merge = merge.dropna(subset=['rating'])
    if merge.empty:
        return None
    # clip scores for stability
    if 'score' in merge.columns:
        loss = float(np.mean((merge['score'] - merge['rating']) ** 2))
        return loss
    return None

# 5) Simple fallback MF using SGD for implicit data

def train_sgd_fallback(train_df, user_to_idx, item_to_idx, seed, F=F, epochs=EPH, lr=LR, reg=REG, val_ratio=VAL_RATIO):
    n_users = len(user_to_idx)
    n_items = len(item_to_idx)
    rng = np.random.default_rng(seed)
    U = rng.normal(scale=0.1, size=(n_users, F))
    V = rng.normal(scale=0.1, size=(n_items, F))

    # Build training triples (u_idx, i_idx, conf)
    triples = []
    for _, row in train_df.iterrows():
        u = user_to_idx.get(int(row['user']))
        i = item_to_idx.get(int(row['item']))
        if u is None or i is None:
            continue
        conf = float(row.get('confidence', 1.0))
        triples.append((u, i, conf))
    if len(triples) == 0:
        return U, V, [], []  # nothing to train

    # Create a small validation set from training triples
    rng.shuffle(triples)
    val_size = max(1, int(val_ratio * len(triples)))
    val_triples = triples[:val_size]
    train_triples = triples[val_size:]
    train_losses = []
    val_losses = []

    for epoch in range(epochs):
        # shuffle training triples
        rng.shuffle(train_triples)
        train_loss = 0.0
        for (u, i, c) in train_triples:
            pred = U[u].dot(V[i])
            e = c - pred
            # SGD updates
            U_old = U[u].copy()
            U[u] += lr * (e * V[i] - reg * U[u])
            V[i] += lr * (e * U_old - reg * V[i])
            train_loss += e * e
        train_loss /= max(1, len(train_triples))
        train_losses.append(train_loss)
        # validation loss
        if len(val_triples) > 0:
            tot = 0.0
            for (u, i, c) in val_triples:
                pred = U[u].dot(V[i])
                e = c - pred
                tot += e * e
            val_loss = tot / len(val_triples)
        else:
            val_loss = 0.0
        val_losses.append(val_loss)
        # print per epoch validation loss
        print(f'  Seed up to epoch {epoch+1}/{epochs}: val_loss={val_loss:.6f}')

    return U, V, train_losses, val_losses

def build_id_maps(interactions):
    users = sorted(interactions['user'].unique())

    user_to_idx = {u: i for i, u in enumerate(users)}
    item_to_idx = {it: i for i, it in enumerate(items)}
    idx_to_user = {i: u for u, i in user_to_idx.items()}
    idx_to_item = {i: it for it, i in item_to_idx.items()}
    return users, items, user_to_idx, item_to_idx, idx_to_user, idx_to_item


def compute_ndcg_and_recall_for_test(test_user_item, top_k_by_user):
    ndcg = 0.0
    recall = 0.0
    for user, true_item in test_user_item.items():
        ranked = top_k_by_user.get(user, [])
        if true_item in ranked:
            rank = ranked.index(true_item) + 1
            ndcg += 1.0 / np.log2(rank + 1)
            recall += 1.0
        else:
            ndcg += 0.0
            recall += 0.0
    n_users = len(test_user_item)
    if n_users == 0:
        return 0.0, 0.0
    return ndcg / n_users, recall / n_users

def sigmoid(x):

    df['item'] = df['item'].astype(int) - 1

# Train/test split by user
def split_by_user(df, test_frac, seed):
    rng = np.random.default_rng(seed)
    trains = []
    tests = []
    for u, g in df.groupby('user'):
        g = g.sample(frac=1.0, random_state=rng.integers(1e9))  # shuffle
        m = len(g)
        if m == 0:
            continue
        n_test = max(1, int(np.ceil(m * test_frac)))
        test_g = g.iloc[:n_test]
        train_g = g.iloc[n_test:]
        if not train_g.empty:
            trains.append(train_g)
        if not test_g.empty:
            tests.append(test_g)
    train_df = pd.concat(trains, ignore_index=True) if trains else df.iloc[0:0].copy()
    test_df = pd.concat(tests, ignore_index=True) if tests else df[0:0].copy()
    return train_df, test_df

# Simple implicit MF (lightweight SGD-based implementation) - serves as a robust baseline
class SimpleImplicitMF:
    def __init__(self, num_users, num_items, latent, lr, reg, seed=None):
        rng = np.random.default_rng(seed)
        self.num_users = int(num_users)
        self.num_items = int(num_items)
        self.latent = int(latent)
        self.lr = float(lr)
        self.reg = float(reg)
        self.U = rng.normal(0, 0.1, size=(self.num_users, self.latent))
        self.V = rng.normal(0, 0.1, size=(self.num_items, self.latent))
    def train_batch(self, us, is_, batch_scores=None):
        # us, is_ are 1D arrays of indices into users/items
        u_batch = np.asarray(us, dtype=np.int32)
        i_batch = np.asarray(is_, dtype=np.int32)
        Uu = self.U[u_batch]  # (B, latent)
        Vi = self.V[i_batch]  # (B, latent)
        pred = np.sum(Uu * Vi, axis=1)
        p = sigmoid(pred)
        err = p - 1.0  # since y=1 for all observed interactions
        # Gradients and updates (vectorized)
        self.U[u_batch] = Uu - self.lr * (err[:, None] * Vi + self.reg * Uu)
        self.V[i_batch] = Vi - self.lr * (err[:, None] * Uu + self.reg * Vi)
        # compute logistic loss contribution for this batch
        loss = -np.log(np.clip(p, 1e-12, 1.0)).mean()
        return loss
    def predict_scores(self, u):
        # return scores for user u over all items
        return self.U[u] @ self.V.T  # (num_items,)

def split_by_user(df, test_frac, seed):
    rng = np.random.default_rng(seed)
    trains = []
    tests = []
    for u, g in df.groupby('user'):

        if not test_g.empty:
            tests.append(test_g)

    test_df = pd.concat(tests, ignore_index=True) if tests else df[0:0].copy()

# Load data
def load_movielens_implicit(path):
    df = pd.read_csv(path, sep='\t', header=None, names=['user','item','rating','ts'])
    df['user'] = df['user'].astype(int) - 1
    df['item'] = df['item'].astype(int) - 1
    # implicit feedback encoded as confidence values for training
    df['interaction'] = (df['rating'] > 0).astype(float)
    # compute confidence: c = 1 + alpha * r where r in {0,1}
    df['confidence'] = 1.0 + ALPHA * df['interaction']
    return df[['user','item','confidence']].copy()

# Split by user with deterministic seed
def split_by_user_implicit(df, test_frac, seed):
    rng = np.random.default_rng(seed)
    trains = []
    tests = []
    for u, g in df.groupby('user'):
        g = g.sample(frac=1.0, random_state=rng.integers(1e9))  # shuffle rows per user
        m = len(g)
        n_test = max(1, int(np.ceil(m * test_frac)))
        test_g = g.iloc[:n_test]
        train_g = g.iloc[n_test:]
        if not train_g.empty:
            trains.append(train_g)
        if not test_g.empty:
            tests.append(test_g)
    train_df = pd.concat(trains, ignore_index=True) if trains else df.iloc[0:0].copy()
    test_df = pd.concat(tests, ignore_index=True) if tests else df.iloc[0:0].copy()
    return train_df, test_df
